### 2.3 Transaction / 事务层

实现 GPT 建议的“Snapshot → Execute → Verify → Commit/Rollback”：

```python
# core/transaction.py
import bpy

class TransactionManager:
    def __init__(self):
        self._snapshots = {}
        self._revision = 0

    def begin(self, task_id: str) -> int:
        """开始事务，返回当前 revision"""
        self._revision += 1
        rev = self._revision
        self._snapshots[task_id] = {
            "revision": rev,
            "filepath": None,  # 可选：保存临时 .blend
        }
        bpy.ops.ed.undo_push(message=f"AI Task {task_id} Start")
        return rev

    def commit(self, task_id: str):
        """提交事务"""
        bpy.ops.ed.undo_push(message=f"AI Task {task_id} Complete")
        self._snapshots.pop(task_id, None)

    def rollback(self, task_id: str):
        """回滚到事务开始前"""
        bpy.ops.ed.undo()  # 单次撤销整个 Task
        self._snapshots.pop(task_id, None)

    def check_revision(self, expected: int) -> bool:
        """检查场景是否被人工修改"""
        return self._revision == expected
```

### 2.6 Validation 层

AI 建模完成后自动验证：

```python
# tools/validation.py
import bpy
import bmesh

def validate_mesh(object_name: str) -> dict:
    obj = bpy.data.objects.get(object_name)
    if not obj or obj.type != 'MESH':
        return {"error": "Not a mesh object"}

    bm = bmesh.new()
    bm.from_mesh(obj.data)

    non_manifold = sum(1 for e in bm.edges if not e.is_manifold)
    loose_verts = sum(1 for v in bm.verts if not v.link_edges)
    ngons = sum(1 for f in bm.faces if len(f.verts) > 4)
    degenerate = sum(1 for f in bm.faces if f.calc_area() < 1e-10)

    bm.free()

    return {
        "manifold": non_manifold == 0,
        "non_manifold_edges": non_manifold,
        "loose_vertices": loose_verts,
        "ngons": ngons,
        "degenerate_faces": degenerate,
        "negative_scale": any(s < 0 for s in obj.scale),
        "has_uv": len(obj.data.uv_layers) > 0,
        "has_material": len(obj.data.materials) > 0,
    }

def validate_uv(object_name: str) -> dict:
    obj = bpy.data.objects.get(object_name)
    if not obj or not obj.data.uv_layers:
        return {"error": "No UV layer"}

    uv = obj.data.uv_layers.active
    if not uv:
        return {"error": "No active UV layer"}

    us = [l.uv[0] for l in uv.data]
    vs = [l.uv[1] for l in uv.data]

    return {
        "uv_layer_count": len(obj.data.uv_layers),
        "uv_vertex_count": len(uv.data),
        "u_range": [min(us), max(us)],
        "v_range": [min(vs), max(vs)],
        "outside_0_1": any(u < 0 or u > 1 for u in us) or
                       any(v < 0 or v > 1 for v in vs),
    }
```

### 2.7 Geometry Nodes 模块

**设计原则**：不让 LLM 直接生成 bpy Geometry Node Python 代码，而是先让 LLM 输出结构化的节点图 JSON，经校验器验证后再转换为 Blender 节点树。

```python
# tools/geometry_nodes.py

# 节点图 JSON Schema 示例
GN_GRAPH_SCHEMA = {
    "type": "object",
    "properties": {
        "nodes": {
            "type": "array",
            "items": {
                "type": "object",
                "properties": {
                    "id": {"type": "string"},
                    "type": {"type": "string"},  # e.g. "GeometryNodeDistributePointsOnFaces"
                    "position": {"type": "array", "items": {"type": "number"}},
                    "inputs": {"type": "object"}
                }
            }
        },
        "links": {
            "type": "array",
            "items": {
                "type": "object",
                "properties": {
                    "from_node": {"type": "string"},
                    "from_socket": {"type": "string"},
                    "to_node": {"type": "string"},
                    "to_socket": {"type": "string"}
                }
            }
        }
    }
}

def build_node_tree(graph_json: dict, group_name: str) -> str:
    """将结构化 JSON 转换为 Blender Geometry Node Group"""
    group = bpy.data.node_groups.new(group_name, 'GeometryNodeTree')

    # 创建节点
    node_map = {}
    for node_def in graph_json["nodes"]:
        node = group.nodes.new(node_def["type"])
        node.location = node_def.get("position", [0, 0])
        node_map[node_def["id"]] = node

    # 建立连接
    for link in graph_json["links"]:
        from_node = node_map[link["from_node"]]
        to_node = node_map[link["to_node"]]
        from_socket = from_node.outputs.get(link["from_socket"])
        to_socket = to_node.inputs.get(link["to_socket"])
        if from_socket and to_socket:
            group.links.new(from_socket, to_socket)

    return group.name
```

### 2.8 MCP Server（独立进程）

完整架构中，MCP Server 作为独立进程运行，通过 TCP Socket 与 Blender 插件通信。Socket 使用 newline-delimited JSON，所有 `bpy` 调用通过队列 + timer 路由到主线程。

```python
# mcp_server/server.py
import json
import socket
from mcp.server import Server
from mcp.types import Tool, TextContent

app = Server("blender-deepseek-agent")

@app.list_tools()
async def list_tools():
    """返回 Blender 插件注册的所有工具"""
    schemas = load_tool_schemas()  # 从插件同步
    return [Tool(**s) for s in schemas]

@app.call_tool()
async def call_tool(name: str, arguments: dict):
    """转发到 Blender 插件的 TCP Socket"""
    result = send_to_blender(name, arguments)
    return [TextContent(type="text", text=json.dumps(result))]

def send_to_blender(tool_name: str, args: dict) -> dict:
    """通过 TCP 发送到 Blender 插件"""
    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.connect(("localhost", 9876))
    payload = json.dumps({"tool": tool_name, "args": args}) + "\n"
    sock.sendall(payload.encode())
    response = b""
    while not response.endswith(b"\n"):
        response += sock.recv(4096)
    sock.close()
    return json.loads(response)
```

### 2.9 线程路由

Blender 的 `bpy` 不是线程安全的。Socket 接收线程必须将操作路由到主线程：

```python
# core/thread_router.py
import bpy
import queue

_command_queue = queue.Queue()

def submit_to_main_thread(func, *args, **kwargs):
    """提交任务到主线程队列"""
    result_holder = {"done": False, "result": None, "error": None}

    def wrapper():
        try:
            result_holder["result"] = func(*args, **kwargs)
        except Exception as e:
            result_holder["error"] = e
        result_holder["done"] = True
        return None  # timer 不重复

    _command_queue.put(wrapper)
    return result_holder

def _process_queue():
    """在 bpy.app.timers 中周期调用"""
    while not _command_queue.empty():
        task = _command_queue.get_nowait()
        task()
    return 0.05  # 50ms 后再次调用

def register():
    bpy.app.timers.register(_process_queue, persistent=True)
```

### 2.10 Async Job 管理

长时间操作（渲染、烘焙、AI 3D 生成）不能阻塞 UI：

```python
# core/job.py
import bpy
import uuid

class JobManager:
    def __init__(self):
        self.jobs = {}

    def create(self, job_type: str, params: dict) -> str:
        job_id = str(uuid.uuid4())[:8]
        self.jobs[job_id] = {
            "id": job_id,
            "type": job_type,
            "status": "pending",
            "progress": 0.0,
            "result": None,
            "params": params,
        }
        return job_id

    def update(self, job_id: str, progress: float):
        if job_id in self.jobs:
            self.jobs[job_id]["progress"] = progress

    def complete(self, job_id: str, result):
        if job_id in self.jobs:
            self.jobs[job_id]["status"] = "completed"
            self.jobs[job_id]["result"] = result
            self.jobs[job_id]["progress"] = 1.0

    def cancel(self, job_id: str):
        if job_id in self.jobs:
            self.jobs[job_id]["status"] = "cancelled"
```

### 2.12 Memory 系统

```python
# agent/memory.py
import json
import os
import bpy

class AgentMemory:
    def __init__(self):
        self.global_memory = {}
        self.project_memory = {}
        self._load()

    def _memory_path(self):
        return os.path.join(
            bpy.utils.user_resource('CONFIG'),
            "blender_deepseek_agent",
            "memory.json"
        )

    def _load(self):
        path = self._memory_path()
        if os.path.exists(path):
            with open(path) as f:
                data = json.load(f)
                self.global_memory = data.get("global", {})
                self.project_memory = data.get("project", {})

    def save(self):
        os.makedirs(os.path.dirname(self._memory_path()), exist_ok=True)
        with open(self._memory_path(), "w") as f:
            json.dump({
                "global": self.global_memory,
                "project": self.project_memory,
            }, f, ensure_ascii=False, indent=2)

    def set_naming_rule(self, category: str, pattern: str):
        """设置命名规则，Agent 后续自动应用"""
        self.project_memory.setdefault("naming_rules", {})[category] = pattern
        self.save()

    def get_naming_rule(self, category: str) -> str:
        return self.project_memory.get("naming_rules", {}).get(category, "")
```

# tools/file_io.py
import bpy

EXPORT_FORMATS = {
    "fbx": {"op": bpy.ops.export_scene.fbx, "ext": ".fbx"},
    "obj": {"op": bpy.ops.wm.obj_export, "ext": ".obj"},
    "glb": {"op": bpy.ops.export_scene.gltf, "ext": ".glb"},
    "stl": {"op": bpy.ops.export_mesh.stl, "ext": ".stl"},

def export_model(object_names: list, format: str, output_path: str) -> dict:
    """导出指定物体"""
    fmt = EXPORT_FORMATS.get(format)
    if not fmt:
        return {"error": f"Unsupported format: {format}"}

    # 选择要导出的物体
    bpy.ops.object.select_all(action='DESELECT')
    for name in object_names:
        obj = bpy.data.objects.get(name)
        if obj:
            obj.select_set(True)

    # 执行导出
    fmt["op"](filepath=output_path, use_selection=True)

    return {
        "status": "ok",
        "format": format,
        "path": output_path,
        "object_count": len(object_names),
    }

def import_model(filepath: str) -> dict:
    """导入模型文件"""
    ext = filepath.rsplit(".", 1)[-1].lower()
    if ext == "fbx":
        bpy.ops.import_scene.fbx(filepath=filepath)
    elif ext == "obj":
        bpy.ops.wm.obj_import(filepath=filepath)
    elif ext in ("glb", "gltf"):
        bpy.ops.import_scene.gltf(filepath=filepath)
    elif ext == "stl":
        bpy.ops.import_mesh.stl(filepath=filepath)
    else:
        return {"error": f"Unsupported import format: {ext}"}

    # 返回导入结果
    imported = [o.name for o in bpy.context.selected_objects]
    return {"status": "ok", "imported": imported}
